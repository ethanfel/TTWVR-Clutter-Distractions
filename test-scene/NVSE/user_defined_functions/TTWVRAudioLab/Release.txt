ref subjectRef
ref homeMarkerRef
float separation
begin Function {subjectRef homeMarkerRef}
 subjectRef.Enable
 let separation := subjectRef.GetDistance homeMarkerRef
 if separation > 6
  subjectRef.MoveTo homeMarkerRef
 endif
 subjectRef.SetAngle Z 0
 if subjectRef.GetDead
  subjectRef.Resurrect 1
 endif
 subjectRef.RestoreAV Health 1000
end
